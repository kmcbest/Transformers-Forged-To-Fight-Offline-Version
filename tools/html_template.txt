    <script>
        const PUA_LIST = __PUA_JSON__;
        const STORAGE_KEY = 'tftf_icon_pua_mappings_v1';

        // Load storage
        let storageData = {};
        try {
            const raw = localStorage.getItem(STORAGE_KEY);
            if (raw) {
                storageData = JSON.parse(raw);
            }
        } catch(e) {
            console.error('Failed to load localStorage', e);
        }

        const gridContainer = document.getElementById('gridContainer');
        const statsBadge = document.getElementById('statsBadge');
        const searchInput = document.getElementById('searchInput');
        const filterFilledBtn = document.getElementById('filterFilledBtn');
        const exportBtn = document.getElementById('exportBtn');
        const importBtn = document.getElementById('importBtn');
        const fileInput = document.getElementById('fileInput');
        const toast = document.getElementById('toast');

        let onlyFilled = false;

        function showToast(msg) {
            toast.textContent = msg;
            toast.style.display = 'block';
            setTimeout(() => {
                toast.style.display = 'none';
            }, 2000);
        }

        function saveToStorage() {
            localStorage.setItem(STORAGE_KEY, JSON.stringify(storageData));
            updateStats();
        }

        function updateStats() {
            let filledCount = 0;
            for (const item of PUA_LIST) {
                const rec = storageData[item.unicode_escape];
                if (rec && (rec.name_en || rec.name_zh)) {
                    filledCount++;
                }
            }
            statsBadge.textContent = `已标注: ${filledCount} / ${PUA_LIST.length}`;
        }

        function renderCards() {
            gridContainer.innerHTML = '';
            const filterText = searchInput.value.trim().toLowerCase();

            for (const item of PUA_LIST) {
                const rec = storageData[item.unicode_escape] || { name_en: '', name_zh: '' };
                const hasFilled = !!(rec.name_en || rec.name_zh);

                if (onlyFilled && !hasFilled) continue;

                if (filterText) {
                    const matchHex = item.hex.toLowerCase().includes(filterText);
                    const matchEscape = item.unicode_escape.toLowerCase().includes(filterText);
                    const matchGlyph = item.glyph_name.toLowerCase().includes(filterText);
                    const matchEn = (rec.name_en || '').toLowerCase().includes(filterText);
                    const matchZh = (rec.name_zh || '').toLowerCase().includes(filterText);
                    if (!matchHex && !matchEscape && !matchGlyph && !matchEn && !matchZh) {
                        continue;
                    }
                }

                const card = document.createElement('div');
                card.className = `card ${hasFilled ? 'is-filled' : ''}`;
                card.dataset.escape = item.unicode_escape;

                card.innerHTML = `
                    <div class="icon-row">
                        <div class="icon-preview-box">
                            <span class="game-icon">${item.char}</span>
                        </div>
                        <div class="icon-meta">
                            <span class="code-pill">${item.unicode_escape}</span>
                            <span class="glyph-name">Glyph: ${item.glyph_name}</span>
                            <span class="glyph-name">Hex: 0x${item.hex} (Dec: ${item.code})</span>
                        </div>
                    </div>
                    <div class="inputs-row">
                        <div class="field-group">
                            <label>EN</label>
                            <input type="text" class="text-input input-en" placeholder="English name / skill..." value="${rec.name_en ? escapeHtml(rec.name_en) : ''}">
                        </div>
                        <div class="field-group">
                            <label>中文</label>
                            <input type="text" class="text-input input-zh" placeholder="中文名 / 技能说明..." value="${rec.name_zh ? escapeHtml(rec.name_zh) : ''}">
                        </div>
                    </div>
                `;

                const inputEn = card.querySelector('.input-en');
                const inputZh = card.querySelector('.input-zh');

                const handleInputChange = () => {
                    const valEn = inputEn.value.trim();
                    const valZh = inputZh.value.trim();
                    if (valEn || valZh) {
                        storageData[item.unicode_escape] = {
                            hex: item.hex,
                            glyph_name: item.glyph_name,
                            name_en: valEn,
                            name_zh: valZh
                        };
                        card.classList.add('is-filled');
                    } else {
                        delete storageData[item.unicode_escape];
                        card.classList.remove('is-filled');
                    }
                    saveToStorage();
                };

                inputEn.addEventListener('input', handleInputChange);
                inputZh.addEventListener('input', handleInputChange);

                gridContainer.appendChild(card);
            }
        }

        function escapeHtml(str) {
            return str.replace(/&/g, '&amp;')
                      .replace(/</g, '&lt;')
                      .replace(/>/g, '&gt;')
                      .replace(/"/g, '&quot;')
                      .replace(/'/g, '&#039;');
        }

        searchInput.addEventListener('input', () => {
            renderCards();
        });

        filterFilledBtn.addEventListener('click', () => {
            onlyFilled = !onlyFilled;
            filterFilledBtn.textContent = onlyFilled ? '显示全部' : '仅看已标注';
            if (onlyFilled) {
                filterFilledBtn.style.background = '#21262d';
                filterFilledBtn.style.borderColor = 'var(--accent)';
            } else {
                filterFilledBtn.style.background = 'transparent';
                filterFilledBtn.style.borderColor = 'var(--card-border)';
            }
            renderCards();
        });

        // Export JSON
        exportBtn.addEventListener('click', () => {
            const exportResult = {};
            for (const item of PUA_LIST) {
                const rec = storageData[item.unicode_escape];
                if (rec && (rec.name_en || rec.name_zh)) {
                    exportResult[item.unicode_escape] = {
                        name_en: rec.name_en || "",
                        name_zh: rec.name_zh || "",
                        glyph_name: item.glyph_name,
                        hex: item.hex
                    };
                }
            }

            const jsonStr = JSON.stringify(exportResult, null, 2);
            const blob = new Blob([jsonStr], { type: 'application/json;charset=utf-8' });
            const url = URL.createObjectURL(blob);
            const a = document.createElement('a');
            a.href = url;
            a.download = 'tftf_font_pua_skills.json';
            document.body.appendChild(a);
            a.click();
            document.body.removeChild(a);
            URL.revokeObjectURL(url);
            showToast('已成功导出 JSON 文件！');
        });

        // Import JSON
        importBtn.addEventListener('click', () => {
            fileInput.click();
        });

        fileInput.addEventListener('change', (e) => {
            const file = e.target.files[0];
            if (!file) return;
            const reader = new FileReader();
            reader.onload = (event) => {
                try {
                    const imported = JSON.parse(event.target.result);
                    for (const [key, val] of Object.entries(imported)) {
                        let k = key;
                        if (!k.startsWith('\\\\u') && !k.startsWith('\\u')) {
                            if (k.startsWith('U+') || k.startsWith('u+')) {
                                k = '\\\\u' + k.slice(2).toLowerCase();
                            }
                        }
                        storageData[k] = {
                            name_en: val.name_en || val.en || "",
                            name_zh: val.name_zh || val.zh || "",
                            glyph_name: val.glyph_name || "",
                            hex: val.hex || ""
                        };
                    }
                    saveToStorage();
                    renderCards();
                    showToast('成功导入并合并数据！');
                } catch (err) {
                    alert('导入失败，请检查 JSON 格式：' + err.message);
                }
            };
            reader.readAsText(file);
            fileInput.value = '';
        });

        // Initial setup
        updateStats();
        renderCards();
    </script>
