        async function fetchFromCloud() {
            setCloudStatus('syncing', I18N[currentLang].cloudSyncing);
            const controller = new AbortController();
            const timeoutId = setTimeout(() => controller.abort(), 4500);

            try {
                const resp = await fetch(`${UPSTASH_CONFIG.url}/hgetall/${UPSTASH_CONFIG.hashKey}`, {
                    headers: {
                        "Authorization": `Bearer ${UPSTASH_CONFIG.token}`

                    signal: controller.signal

                const data = await resp.json();
                if (data && Array.isArray(data.result)) {

                            cloudMap[key] = JSON.parse(data.result[i + 1]);
                        } catch (e) {
                            cloudMap[key] = { name_en: data.result[i + 1], name_zh: "" };
                        }

                        storageData[k] = {
                            hex: v.hex || storageData[k]?.hex || "",
                            glyph_name: v.glyph_name || storageData[k]?.glyph_name || "",
                            name_en: v.name_en || "",
                            name_zh: v.name_zh || "",
                            author: v.author || "Community",
                            updated_at: v.updated_at || ""
                        };
                    }
                    saveToStorage();
                    const count = Object.keys(cloudMap).length;
                    setCloudStatus('online', `${I18N[currentLang].cloudOnline} (${count})`);
                } else {
                    setCloudStatus('online', `${I18N[currentLang].cloudOnline} (0)`);
                }
            } catch (err) {
                clearTimeout(timeoutId);
                console.warn("Upstash fetch notice (using local cache):", err.message || err);
                setCloudStatus('offline', I18N[currentLang].cloudOffline);
            } finally {
                renderCards();
                updateStats();
            }
        }

                            name_en: valEn,
                            name_zh: valZh,
                            author: getNickname(),
                            updated_at: new Date().toISOString()
                        };
                        storageData[esc] = recData;
                        card.classList.add('is-filled');

                        // Debounce sync to Upstash
                        clearTimeout(saveTimeoutMap[esc]);
                        saveTimeoutMap[esc] = setTimeout(() => {
                            syncKeyToCloud(esc, recData);
                        }, 700);
                    } else {
                        delete storageData[esc];
                        card.classList.remove('is-filled');

                        clearTimeout(saveTimeoutMap[esc]);
                        saveTimeoutMap[esc] = setTimeout(() => {
                            syncKeyToCloud(esc, null);
                        }, 700);
                    }
                    saveToStorage();
                };

                inputEn.addEventListener('input', handleInputChange);
                inputZh.addEventListener('input', handleInputChange);

                gridContainer.appendChild(card);
            }
        }

        function escapeHtml(str) {
            return str.replace(/&/g, '&amp;')
                      .replace(/</g, '&lt;')
                      .replace(/>/g, '&gt;')
                      .replace(/"/g, '&quot;')
                      .replace(/'/g, '&#039;');
        }

        searchInput.addEventListener('input', () => {
            renderCards();
        });

        filterFilledBtn.addEventListener('click', () => {
            onlyFilled = !onlyFilled;
            filterFilledBtn.textContent = onlyFilled ? I18N[currentLang].filterAll : I18N[currentLang].filterFilled;
            if (onlyFilled) {
                filterFilledBtn.style.background = '#21262d';
                filterFilledBtn.style.borderColor = 'var(--accent)';
            } else {
                filterFilledBtn.style.background = 'transparent';
                filterFilledBtn.style.borderColor = 'var(--card-border)';
            }
            renderCards();
        });

        // Export JSON
        exportBtn.addEventListener('click', () => {
            const exportResult = {};
            for (const item of PUA_LIST) {
                const rec = storageData[item.unicode_escape];
                if (rec && (rec.name_en || rec.name_zh)) {
                    exportResult[item.unicode_escape] = {
                        name_en: rec.name_en || "",
                        name_zh: rec.name_zh || "",
                        glyph_name: item.glyph_name,
                        hex: item.hex,
                        author: rec.author || ""
                    };
                }
            }

            const jsonStr = JSON.stringify(exportResult, null, 2);
            const blob = new Blob([jsonStr], { type: 'application/json;charset=utf-8' });
            const url = URL.createObjectURL(blob);
            const a = document.createElement('a');
            a.href = url;
            a.download = 'tftf_font_pua_skills.json';
            document.body.appendChild(a);
            a.click();
            document.body.removeChild(a);
            URL.revokeObjectURL(url);
            showToast(I18N[currentLang].exportSuccess);
        });

        // Import JSON
        importBtn.addEventListener('click', () => {
            fileInput.click();
        });

        fileInput.addEventListener('change', (e) => {
            const file = e.target.files[0];
            if (!file) return;
            const reader = new FileReader();
            reader.onload = async (event) => {
                try {
                    const imported = JSON.parse(event.target.result);
                    let importedCount = 0;
                    for (const [key, val] of Object.entries(imported)) {
                        let k = key;
                        if (!k.startsWith('\\u') && !k.startsWith('\u')) {
                            if (k.startsWith('U+') || k.startsWith('u+')) {
                                k = '\\u' + k.slice(2).toLowerCase();
                            }
                        }
                        const rec = {
                            name_en: val.name_en || val.en || "",
                            name_zh: val.name_zh || val.zh || "",
                            glyph_name: val.glyph_name || "",
                            hex: val.hex || "",
                            author: val.author || getNickname(),
                            updated_at: new Date().toISOString()
                        };
                        storageData[k] = rec;
                        syncKeyToCloud(k, rec);
                        importedCount++;
                    }
                    saveToStorage();
                    renderCards();
                    showToast(`${I18N[currentLang].importSuccess} (${importedCount})`);
                } catch (err) {
                    alert(I18N[currentLang].importFail + err.message);
                }
            };
            reader.readAsText(file);
            fileInput.value = '';
        });

        // Initialize
        applyI18n();
        renderCards();
        fetchFromCloud();
    </script>
